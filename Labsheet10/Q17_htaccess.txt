# Q17: public/.htaccess
# Clean URLs + force HTTPS + disable directory listing

Options -Indexes
DirectoryIndex index.php

<IfModule mod_rewrite.c>
    RewriteEngine On

    # --- Force HTTPS ---
    RewriteCond %{HTTPS} off
    RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # --- Pass Authorization header to PHP (for token auth, Q8) ---
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # --- Clean URLs: send everything that is not a real file/dir to index.php ---
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^(.*)$ index.php [QSA,L]
</IfModule>

# Custom error pages
ErrorDocument 404 /errors/404.html
ErrorDocument 500 /errors/500.html

# Block access to sensitive files
<FilesMatch "\.(env|log|ini|sql|sh|md)$">
    Require all denied
</FilesMatch>

# Security headers (needs mod_headers)
<IfModule mod_headers.c>
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
</IfModule>

# Required Apache modules:  sudo a2enmod rewrite headers && sudo systemctl restart apache2
# Virtual host must contain: AllowOverride All
